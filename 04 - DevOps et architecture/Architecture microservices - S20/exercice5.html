Contrôleur

src/infrastructure/controllers/appointmentController.js 
const Appointment = require('../../domain/aggregates/Appointment');
const { v4: uuidv4 } = require('uuid');
const appointmentQueue = require('../queue/appointmentQueue');

class AppointmentController {
  
  // POST /appointments - Create appointment
  async create(req, res) {
    try {
      const { patientId, doctorId, dateTime, specialty } = req.body;
      
      // Validation métier
      if (!patientId || !doctorId || !dateTime || !specialty) {
        return res.status(400).json({
          error: 'Missing required fields: patientId, doctorId, dateTime, specialty'
        });
      }

      // Vérifier que la date est future
      const appointmentDate = new Date(dateTime);
      if (appointmentDate <= new Date()) {
        return res.status(400).json({
          error: 'Appointment date must be in the future'
        });
      }

      const appointment = new Appointment(
        uuidv4(),
        patientId,
        doctorId,
        dateTime,
        specialty
      );

      // Publier l'événement vers RabbitMQ
      await appointmentQueue.publishAppointmentScheduled(appointment.toJSON());

      res.status(201).json({
        message: 'Appointment created successfully',
        appointment: appointment.toJSON()
      });

    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }

  // GET /appointments/:id - Get appointment by ID
  async getById(req, res) {
    try {
      const { id } = req.params;
      
      const appointment = {
        id,
        patientId: 'pat-001',
        doctorId: 'dr-001',
        dateTime: new Date('2024-02-15T14:00:00Z'),
        specialty: 'Cardiologie',
        status: 'SCHEDULED'
      };

      if (!appointment) {
        return res.status(404).json({ error: 'Appointment not found' });
      }

      res.json(appointment);

    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }

  // PUT /appointments/:id/cancel - Cancel appointment
  async cancel(req, res) {
    try {
      const { id } = req.params;
      
      const appointment = new Appointment(
        id,
        'pat-001',
        'dr-001',
        '2024-02-15T14:00:00Z',
        'Cardiologie'
      );

      appointment.cancel();

      await appointmentQueue.publishAppointmentCancelled(appointment.toJSON());

      res.json({
        message: 'Appointment cancelled successfully',
        appointment: appointment.toJSON()
      });

    } catch (error) {
      res.status(400).json({ error: error.message });
    }
  }
}

module.exports = new AppointmentController();

Service avec Transactions

notification-service/src/services/notificationService.js 
const mongoose = require('mongoose');

class NotificationService {
  
  // Transaction pour confirmation de rendez-vous
  async sendConfirmation(appointment) {
    const session = await mongoose.startSession();
    
    try {
      session.startTransaction();
      
      console.log(`[Notification] Starting transaction: ${appointment.id}`);
      
      const notification = {
        id: `notif-${Date.now()}`,
        type: 'EMAIL',
        recipient: appointment.patientId,
        subject: `Confirmation RDV - ${appointment.specialty}`,
        message: `Votre rendez-vous est confirmé`,
        status: 'PENDING',
        createdAt: new Date()
      };

      // Sauvegarder notification
      // await NotificationModel.create(notification);
      
      // Simuler envoi email
      notification.status = 'SENT';
      
      // Valider transaction
      await session.commitTransaction();
      session.endSession();
      
      console.log(`[Notification] Transaction committed`);
      return notification;
      
    } catch (error) {
      // Annuler transaction en cas d'erreur
      await session.abortTransaction();
      session.endSession();
      console.error(`[Notification] Transaction failed:`, error.message);
      throw error;
    }
  }

  // sendCancellation et sendCompletion similaires avec transactions...
}

module.exports = new NotificationService();

Point d'entrée (index.js)

notification-service/src/index.js 
require('dotenv').config();
const express = require('express');
const notificationRoutes = require('./infrastructure/controllers/notificationRoutes');
const appointmentConsumer = require('./infrastructure/queue/appointmentConsumer');
const db = require('./infrastructure/database/mongodb');

const app = express();
const PORT = process.env.PORT || 3006;

// Middleware
app.use(express.json());

// Health check
app.get('/health', (req, res) => {
  res.json({ status: 'OK', service: 'notification-service' });
});

// Routes
app.use('/api/notifications', notificationRoutes);

// Start server
app.listen(PORT, async () => {
  console.log(`Notification Service running on port ${PORT}`);
  
  // Connect to RabbitMQ and start consuming
  await appointmentConsumer.connect();
  
  // Connect to MongoDB
  await db.connect();
});

module.exports = app;
